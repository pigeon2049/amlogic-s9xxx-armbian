cmake_minimum_required(VERSION 3.16)
project(bluedevil_wizard_backport VERSION 6.3.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(KDE_COMPILERSETTINGS_LEVEL "5.82")
find_package(ECM 6.10 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
find_package(Qt6 6.7 REQUIRED COMPONENTS Core DBus Widgets)
find_package(KF6 6.10 REQUIRED COMPONENTS
    BluezQt CoreAddons DBusAddons I18n Notifications WidgetsAddons WindowSystem)
include(KDECMakeSettings)
include(KDECompilerSettings NO_POLICY_SCOPE)
include(ECMQtDeclareLoggingCategory)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h"
    "#define BLUEDEVIL_VERSION_STRING \"6.3.4-w103d1\"\n")
include_directories("${CMAKE_CURRENT_BINARY_DIR}")
add_compile_definitions(TRANSLATION_DOMAIN="bluedevil")
if(NOT IS_DIRECTORY "${BLUEDEVIL_SOURCE}/src/wizard")
    message(FATAL_ERROR "BLUEDEVIL_SOURCE must contain patched BlueDevil 6.3.4 sources")
endif()
add_subdirectory("${BLUEDEVIL_SOURCE}/src/wizard" wizard)
